ALA — Autonomous Life Assistant

An AI-powered care platform built to help elderly users live independently. It watches for falls in real time, routes context to an AI layer for triage, and escalates to caregivers automatically when something looks wrong — no one has to notice in time to act.

I built the majority of this full-stack: the Express API, the Firebase real-time data layer, the pluggable AI adapter routing (OpenAI / Gemini / Claude), and the Twilio-based emergency escalation pipeline.

alerts.escalate.ts 
// POST /api/alerts/escalate
// Triggered when the vision model flags a fall event
export async function escalateFallAlert(event: FallEvent) {
  const patient = await db
    .collection("patients")
    .doc(event.patientId)
    .get();

  const severity = classifySeverity(event.confidence);

  await db.collection("alerts").add({
    patientId: event.patientId,
    severity,
    detectedAt: event.timestamp,
    status: "pending_ack",
  });

  if (severity === "high") {
    await notifyCaregiver({
      to: patient.data().caregiverPhone,
      channel: "sms",
      body: `⚠ Fall detected — ${patient.data().name}. 
Confidence ${event.confidence}%. Responding...`,
    });
  }

  return { escalated: true, severity };
}
